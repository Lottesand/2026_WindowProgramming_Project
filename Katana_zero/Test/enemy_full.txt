Enemy::Enemy(float startX, float startY, EnemyType type, float patrolRange) {
    m_startX = startX; m_startY = startY; m_x = startX; m_y = startY;
    m_vx = 2.0f; m_vy = 0.0f; m_colW = 40.0f; m_colH = 60.0f;
    m_isAlive = true; m_isFacingLeft = false; m_Type = type; m_State = EnemyState::IDLE;
    m_CurrentFrame = 0; m_LastTime = GetTickCount(); m_friction = 0.94f; m_knockbackVx = 0.0f;
    m_isImmortal = false; m_patternTimer = GetTickCount(); m_isWaiting = false; m_walkDistance = 0.0f;
    m_detectionRange = 200.0f; m_isPlayerDetected = false; m_patrolRange = patrolRange;
}
Enemy::~Enemy() {}

void Enemy::CheckPlayerDetection(const Player& player) {
    if (!m_isAlive) { m_isPlayerDetected = false; return; }
    float px = player.GetX() + player.GetColW() / 2.0f, py = player.GetY() + player.GetColH() / 2.0f;
    float ex = m_x + m_colW / 2.0f, ey = m_y + m_colH / 2.0f;
    float dx = fabs(px - ex), dy = fabs(py - ey);
    m_isPlayerDetected = (dx <= m_detectionRange && dy <= 50.0f);
}

void Enemy::RenderDebug(HDC hdc, float camX, float camY, float mapScale) {
    int sx = (int)((m_x - camX) * mapScale), sy = (int)((m_y - camY) * mapScale);
    HBRUSH redBrush = CreateSolidBrush(RGB(255, 0, 0));
    RECT rect = { sx, sy, sx + (int)(m_colW * mapScale), sy + (int)(m_colH * mapScale) };
    FrameRect(hdc, &rect, redBrush); DeleteObject(redBrush);
    COLORREF dc = m_isPlayerDetected ? RGB(255, 0, 0) : RGB(150, 0, 0);
    HPEN hP = CreatePen(PS_SOLID, m_isPlayerDetected ? 3 : 1, dc); HPEN hOP = (HPEN)SelectObject(hdc, hP);
    int r = (int)(m_detectionRange * mapScale), cx = sx + (int)(m_colW * mapScale / 2), cy = sy + (int)(m_colH * mapScale / 2);
    MoveToEx(hdc, cx - r, cy, NULL); LineTo(hdc, cx + r, cy);
    SelectObject(hdc, hOP); DeleteObject(hP);
}

    m_isAlive = true; m_isFacingLeft = false; m_State = EnemyState::IDLE;

void Enemy::ApplyKnockback(float vx) { m_knockbackVx = vx; m_vx = vx; }

void Enemy::Update(float ts, const Player& player) {
    // 1. ?섑룊 ?대룞 泥섎━ (?됰갚 ?ы븿)
    if (m_vx != 0.0f) {
        float nx = m_x + m_vx * ts;
        if (!CheckMapCollision(nx, m_y, m_colW, m_colH)) {
            m_x = nx;
        } else {
            m_vx = 0.0f;
            m_knockbackVx = 0.0f;
        }
        // 二쎌뿀???뚮뒗 留덉같???곸슜
        if (!m_isAlive) m_vx *= m_friction;
    }
    
    // ?됰갚 ?섑룊 泥섎━ (湲곗〈 濡쒖쭅 ?좎??섎릺 m_vx? ?듯빀 愿由?沅뚯옣)
    if (fabs(m_knockbackVx) > 0.1f) {
        float nx = m_x + m_knockbackVx * ts;
        if (!CheckMapCollision(nx, m_y, m_colW, m_colH)) m_x = nx;
        else m_knockbackVx = 0.0f;
        m_knockbackVx *= m_friction;
    }

    // 2. ?섏쭅 ?대룞 諛?以묐젰 泥섎━
    float gravity = m_isAlive ? 2.0f : 1.2f;
    m_vy += gravity * ts;
    if (m_vy > 30.0f) m_vy = 30.0f;

    float nextY = m_y + m_vy;
    bool hf = false;
    
    // 異⑸룎 ?먯젙 ?⑥닔 (?뚮젅?댁뼱? ?숈씪???뺣? 濡쒖쭅)
    auto cf = [&](float ty) {
        int tl = GetCollisionType((int)(m_x + 2.0f), (int)(ty + m_colH));
        int tc = GetCollisionType((int)(m_x + m_colW / 2.0f), (int)(ty + m_colH));
        int tr = GetCollisionType((int)(m_x + m_colW - 2.0f), (int)(ty + m_colH));
        
        if (tl == 1 || tl == 3 || tc == 1 || tc == 3 || tr == 1 || tr == 3) return true;
        if (tl == 2 || tc == 2 || tr == 2) {
            if (m_y + m_colH <= ty + m_colH) return true;
        }
        return false;
    };

    if (m_vy > 0) {
        for (float sy = m_y; sy <= nextY; sy += 1.0f) {
            if (cf(sy)) {
                m_y = sy;
                hf = true;
                break;
            }
        }
        if (hf) m_vy = 0;
        else m_y = nextY;
    } else if (m_vy < 0) {
        auto ch = [&](float ty) {
            int tl = GetCollisionType((int)(m_x + 2.0f), (int)ty);
            int tc = GetCollisionType((int)(m_x + m_colW / 2.0f), (int)ty);
            int tr = GetCollisionType((int)(m_x + m_colW - 2.0f), (int)ty);
            return (tl == 1 || tl == 3 || tc == 1 || tc == 3 || tr == 1 || tr == 3);
        };
        bool hc = false;
        for (float sy = m_y; sy >= nextY; sy -= 1.0f) {
            if (ch(sy)) {
                m_y = sy;
                hc = true;
                break;
            }
        }
        if (hc) m_vy = 0;
        else m_y = nextY;
    } else {
        m_y = nextY;
    }
}

void Gangster::Update(float ts, const Player& player) {
    CheckPlayerDetection(player);
    
    // 1. 怨듯넻 臾쇰━ ?낅뜲?댄듃 (以묐젰, ?대룞, 異⑸룎) ?몄텧
    Enemy::Update(ts, player);

    if (!m_isAlive) {
        // 怨듭쨷 遺??以?諛붾떏???우쑝硫?泥섎컯?덈뒗 紐⑥뀡?쇰줈 蹂寃?        if (m_ActionState == GangsterAction::HURT_FLY && m_vy == 0) {
            m_ActionState = GangsterAction::HURT_GROUND;
            m_State = EnemyState::DEAD;
            m_vx = 0;
            m_CurrentFrame = 0;
        }

        if (GetTickCount() - m_LastTime >= 100) {
            if (m_ActionState == GangsterAction::HURT_GROUND || m_State == EnemyState::DEAD) {
                if (m_CurrentFrame < 13) m_CurrentFrame++;
            } else m_CurrentFrame++;
            m_LastTime = GetTickCount();
        }
        return;
    }

    // 2. ?멸났吏??濡쒖쭅 (?앹〈 ??
    if (m_isWaiting) {
        m_vx = 0.0f; m_State = EnemyState::IDLE;
        if (GetTickCount() - m_patternTimer >= 2000) {
            m_isWaiting = false; m_patternTimer = GetTickCount(); m_walkDistance = 0.0f;
            m_isFacingLeft = !m_isFacingLeft; m_vx = m_isFacingLeft ? -2.0f : 2.0f;
        }
    } else {
        m_State = EnemyState::WALK; float nx = m_x + m_vx * ts;
        if (!CheckMapCollision(nx, m_y, m_colW, m_colH)) {
            m_x = nx; m_walkDistance += fabs(m_vx * ts); m_isFacingLeft = (m_vx < 0.0f);
            if (m_walkDistance >= m_patrolRange) { m_isWaiting = true; m_patternTimer = GetTickCount(); }
        } else { m_isWaiting = true; m_patternTimer = GetTickCount(); }
    }
    if (m_vy > 0.1f) m_State = EnemyState::FALL;
    if (GetTickCount() - m_LastTime > 100) { m_CurrentFrame++; m_LastTime = GetTickCount(); }
}

    if (m_isFacingLeft) {
        if (m_ActionState == GangsterAction::HURT_FLY) img = &m_ImgHurtFly_L[m_CurrentFrame % 2];

        else if (m_State == EnemyState::WALK) img = &m_ImgWalk_L[m_CurrentFrame % 8];
    } else {
        if (m_ActionState == GangsterAction::HURT_FLY) img = &m_ImgHurtFly_R[m_CurrentFrame % 2];
        else if (m_ActionState == GangsterAction::HURT_GROUND || m_State == EnemyState::DEAD) img = &m_ImgHurtGround_R[m_CurrentFrame % 14];

        else if (m_State == EnemyState::WALK) img = &m_ImgWalk_R[m_CurrentFrame % 8];
    }
    if (img && !img->IsNull()) {
        int fw = (int)(img->GetWidth() * es * msX * mapScale), fh = (int)(img->GetHeight() * es * msY * mapScale);
        int fy = sy + (int)(m_colH * mapScale) - fh, dx = sx + (int)(m_colW * mapScale / 2) - (fw / 2);
        img->Draw(hdc, dx, fy, fw, fh);
    }
    if (showDebugRect) RenderDebug(hdc, camX, camY, mapScale);
}

void Grunt::Update(float ts, const Player& player) {
    CheckPlayerDetection(player);
    
    // 1. 怨듯넻 臾쇰━ ?낅뜲?댄듃 ?몄텧
    Enemy::Update(ts, player);

    if (!m_isAlive) {
        if (m_ActionState == GruntAction::HURT_FLY && m_vy == 0) {
            m_ActionState = GruntAction::HURT_GROUND;
            m_State = EnemyState::DEAD;
            m_vx = 0;
            m_CurrentFrame = 0;
        }

        if (GetTickCount() - m_LastTime >= 100) {
            if (m_ActionState == GruntAction::HURT_GROUND || m_State == EnemyState::DEAD) {
                if (m_CurrentFrame < 15) m_CurrentFrame++;
            } else m_CurrentFrame++;
            m_LastTime = GetTickCount();
        }
        return;
    }

    // 2. ?멸났吏??濡쒖쭅
    if (m_isWaiting) {
        m_vx = 0.0f; m_State = EnemyState::IDLE;
        if (GetTickCount() - m_patternTimer >= 1500) {
            m_isWaiting = false; m_patternTimer = GetTickCount();
            m_isFacingLeft = !m_isFacingLeft; m_vx = m_isFacingLeft ? -2.5f : 2.5f;
        }
    } else {
        m_State = EnemyState::WALK; float nx = m_x + m_vx * ts;
        bool outOfRange = false;
        if (m_isFacingLeft && nx < m_startX - m_patrolRange / 2.0f) outOfRange = true;
        if (!m_isFacingLeft && nx > m_startX + m_patrolRange / 2.0f) outOfRange = true;

        if (!CheckMapCollision(nx, m_y, m_colW, m_colH) && !outOfRange) {
            m_x = nx; m_isFacingLeft = (m_vx < 0.0f);
        } else { m_isWaiting = true; m_patternTimer = GetTickCount(); }
    }
    if (m_vy > 0.1f) m_State = EnemyState::FALL;
    if (GetTickCount() - m_LastTime > 100) { m_CurrentFrame++; m_LastTime = GetTickCount(); }
}

void Pomp::Update(float ts, const Player& player) {
    CheckPlayerDetection(player);
    
    // 1. 怨듯넻 臾쇰━ ?낅뜲?댄듃 ?몄텧
    Enemy::Update(ts, player);

    if (!m_isAlive) {
        if (m_ActionState == PompAction::HURT_FLY && m_vy == 0) {
            m_ActionState = PompAction::HURT_GROUND;
            m_State = EnemyState::DEAD;
            m_vx = 0;
            m_CurrentFrame = 0;
        }

        if (GetTickCount() - m_LastTime >= 100) {
            if (m_ActionState == PompAction::HURT_GROUND || m_State == EnemyState::DEAD) {
                if (m_CurrentFrame < 14) m_currentFrame++;
            } else m_CurrentFrame++;
            m_LastTime = GetTickCount();
        }
        return;
    }

    // 2. ?멸났吏??濡쒖쭅
    if (m_isWaiting) {
        m_vx = 0.0f; m_State = EnemyState::IDLE;
        if (GetTickCount() - m_patternTimer >= 1800) {
            m_isWaiting = false; m_patternTimer = GetTickCount(); m_walkDistance = 0.0f;
            m_isFacingLeft = !m_isFacingLeft; m_vx = m_isFacingLeft ? -2.2f : 2.2f;
        }
    } else {
        m_State = EnemyState::WALK; float nx = m_x + m_vx * ts;
        if (!CheckMapCollision(nx, m_y, m_colW, m_colH)) {
            m_x = nx; m_walkDistance += fabs(m_vx * ts); m_isFacingLeft = (m_vx < 0.0f);
            if (m_walkDistance >= m_patrolRange) { m_isWaiting = true; m_patternTimer = GetTickCount(); }
        } else { m_isWaiting = true; m_patternTimer = GetTickCount(); }
    }
    if (m_vy > 0.1f) m_State = EnemyState::FALL;
    if (GetTickCount() - m_LastTime > 100) { m_CurrentFrame++; m_LastTime = GetTickCount(); }
}

void ShieldCop::Update(float ts, const Player& player) {
    CheckPlayerDetection(player);
    if (m_ActionState == ShieldCopAction::HURT_FLY || m_ActionState == ShieldCopAction::HURT_GROUND || m_State == EnemyState::DEAD) {
        if (m_ActionState == ShieldCopAction::HURT_FLY) {
            m_vy += 1.5f * ts;
            float nextX = m_x + m_vx * ts;
            float nextY = m_y + m_vy * ts;

            if (!CheckMapCollision(nextX, m_y, m_colW, m_colH)) m_x = nextX;
            else m_vx = 0;

            if (CheckMapCollision(m_x, nextY, m_colW, m_colH)) {
                if (m_vy > 0) {
                    m_ActionState = ShieldCopAction::HURT_GROUND; m_State = EnemyState::DEAD; m_vx = 0; m_vy = 0; m_CurrentFrame = 0;
                    while (CheckMapCollision(m_x, m_y, m_colW, m_colH)) m_y -= 1.0f;
                } else m_vy = 0;
            } else m_y = nextY;

            m_vx *= m_friction;
        }
        if (GetTickCount() - m_LastTime >= 100) {
            if (m_ActionState == ShieldCopAction::HURT_GROUND || m_State == EnemyState::DEAD) {
                if (m_CurrentFrame < 14) m_CurrentFrame++;
            } else m_CurrentFrame++;
            m_LastTime = GetTickCount();
        }
        return;
    }
    Enemy::Update(ts, player);
    if (m_isWaiting) {
        m_vx = 0.0f; m_State = EnemyState::IDLE;
        if (GetTickCount() - m_patternTimer >= 2200) {
            m_isWaiting = false; m_patternTimer = GetTickCount(); m_walkDistance = 0.0f;
            m_isFacingLeft = !m_isFacingLeft; m_vx = m_isFacingLeft ? -1.8f : 1.8f;
        }
    } else {
        m_State = EnemyState::WALK; float nx = m_x + m_vx * ts;
        if (!CheckMapCollision(nx, m_y, m_colW, m_colH)) {
            m_x = nx; m_walkDistance += fabs(m_vx * ts); m_isFacingLeft = (m_vx < 0.0f);
            if (m_walkDistance >= 140.0f) { m_isWaiting = true; m_patternTimer = GetTickCount(); }
        } else { m_isWaiting = true; m_patternTimer = GetTickCount(); }
    }
    if (m_vy > 0.1f) m_State = EnemyState::FALL;
    if (GetTickCount() - m_LastTime > 100) { m_CurrentFrame++; m_LastTime = GetTickCount(); }
}
